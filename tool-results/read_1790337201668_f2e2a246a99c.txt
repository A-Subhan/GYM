    30→export function CoaModule() {
    31→  const { session, has } = useApp()
    32→  const [search, setSearch] = useState('')
    33→  const [type, setType] = useState('all')
    34→  const [expanded, setExpanded] = useState<Set<string>>(new Set())
    35→  const [open, setOpen] = useState(false)
    36→  const [form, setForm] = useState<any>({})
    37→  const { data, reload } = useFetch<any>(`/api/accounts?accountType=${type !== 'all' ? type : ''}`)
    38→
    39→  const accounts = data?.accounts || []
    40→  const byId = new Map(accounts.map((a: any) => [a.id, a]))
    41→  const roots = accounts.filter(a => !a.parentId)
    42→  const matches = (a: any) => !search || a.name.toLowerCase().includes(search.toLowerCase()) || a.code.includes(search)
    43→  const visible = (a: any, ancestors: any[] = []) => {
    44→    if (matches(a)) return true
    45→    return ancestors.some(matches)
    46→  }
    47→
    48→  const toggle = (id: string) => {
    49→    setExpanded(prev => {
    50→      const n = new Set(prev)
    51→      if (n.has(id)) n.delete(id); else n.add(id)
    52→      return n
    53→    })
    54→  }
    55→  const expandAll = () => setExpanded(new Set(accounts.map((a: any) => a.id)))
    56→  const collapseAll = () => setExpanded(new Set())
    57→
    58→  const renderNode = (a: any, depth = 0): ReactNode => {
    59→    const children = accounts.filter((c: any) => c.parentId === a.id)
    60→    const isExpanded = expanded.has(a.id)
    61→    return (
    62→      <div key={a.id}>
    63→        <div
    64→          className={`flex items-center gap-2 px-2 py-1.5 hover:bg-muted/40 cursor-pointer ${matches(a) ? '' : 'opacity-60'}`}
    65→          style={{ paddingLeft: `${depth * 20 + 8}px` }}
    66→          onClick={() => children.length ? toggle(a.id) : null}
    67→        >
    68→          {children.length ? (
    69→            isExpanded ? <ChevronDown className="h-3 w-3 shrink-0" /> : <ChevronRight className="h-3 w-3 shrink-0" />
    70→          ) : <div className="w-3" />}
    71→          <span className="font-mono text-xs text-muted-foreground w-20">{a.code}</span>
    72→          <span className="flex-1 text-sm">{a.name}</span>
    73→          <Badge variant="outline" className="text-xs">{a.accountType}</Badge>
    74→          {a.bookType && <Badge variant="secondary" className="text-xs">{a.bookType}</Badge>}
    75→          {a.isControl && <Badge className="text-xs">Control</Badge>}
    76→          {a.accountTag && <Badge variant="secondary" className="text-xs">{a.accountTag}</Badge>}
    77→          {has('finance.coa') && (
    78→            <button onClick={(e) => { e.stopPropagation(); setForm({ parentId: a.id, accountType: a.accountType }); setOpen(true) }} className="text-xs text-primary hover:underline">
    79→              Add child
    80→            </button>
    81→          )}
    82→        </div>
    83→        {isExpanded && children.map((c: any) => renderNode(c, depth + 1))}
    84→      </div>
    85→    )
    86→  }
    87→
    88→  return (
    89→    <div>
    90→      <PageHeader title="Chart of Accounts"
    91→        action={has('finance.coa') ? () => { setForm({}); setOpen(true) } : undefined}
    92→        actionLabel="Add Account" />
    93→      <Toolbar>
    94→        <SearchInput value={search} onChange={setSearch} placeholder="Search by name or code…" />
    95→        <Select value={type} onValueChange={setType}>
    96→          <SelectTrigger className="w-40"><SelectValue placeholder="Type" /></SelectTrigger>
    97→          <SelectContent>
    98→            <SelectItem value="all">All types</SelectItem>
    99→            <SelectItem value="Asset">Asset</SelectItem>
   100→            <SelectItem value="Liability">Liability</SelectItem>
   101→            <SelectItem value="Equity">Capital / Equity</SelectItem>
   102→            <SelectItem value="Revenue">Revenue</SelectItem>
   103→            <SelectItem value="Expense">Expense</SelectItem>
   104→          </SelectContent>
   105→        </Select>
   106→        <Button variant="outline" size="sm" onClick={expandAll}>Expand All</Button>
   107→        <Button variant="outline" size="sm" onClick={collapseAll}>Collapse All</Button>
   108→        <Button variant="ghost" size="sm" onClick={reload}>Refresh</Button>
   109→      </Toolbar>
   110→      <Card>
   111→        <CardContent className="p-0">
   112→          {accounts.length === 0 ? <EmptyState message="No accounts" /> : roots.map((a: any) => renderNode(a, 0))}
   113→        </CardContent>
   114→      </Card>
   115→
   116→      <AccountFormModal open={open} onClose={() => setOpen(false)} form={form} setForm={setForm}
   117→        onSaved={() => { setOpen(false); reload() }} accounts={accounts} />
   118→    </div>
   119→  )
   120→}

   300→function VoucherFormModal({ open, onClose, defaultType, onSaved }: any) {
   301→  const { session, branches } = useApp()
   302→  const { data: accountsData } = useFetch<any>('/api/accounts')
   303→  const { data: taxHeadsData } = useFetch<any>('/api/tax-heads')
   304→  const accounts = accountsData?.accounts || []
   305→  const taxHeads = taxHeadsData?.taxHeads || []
   306→  const isJV = defaultType === 'JV'
   307→  const isCashType = defaultType === 'CRV' || defaultType === 'CPV'
   308→  const isBankType = defaultType === 'BRV' || defaultType === 'BPV'
   309→  const bookAccounts = accounts.filter((a: any) =>
   310→    a.isActive && (isCashType ? a.bookType === 'Cash' : isBankType ? a.bookType === 'Bank' : false)
   311→  )
   312→  const detailAccounts = accounts.filter((a: any) => a.isDetail && a.isActive)
   313→
   314→  const emptyLine = () => ({
   315→    accountId: '', lineDescription: '', amount: 0, debit: 0, credit: 0,
   316→    taxAccountId: '', taxRate: 0, taxAmount: 0,
   317→    chequeNo: '', chequeAmount: 0, chequeBankName: '', chequeStatus: '',
   318→    status: 'Active',
   319→  })
   320→
   321→  const [form, setForm] = useState<any>({
   322→    voucherDate: new Date().toISOString().slice(0, 10),
   323→    reference: '', bookAccountId: '', branchId: session?.branchId || branches[0]?.id || '',
   324→    description: '', lines: [emptyLine()],
   325→  })
   326→
   327→  useEffect(() => {
   328→    if (open) {
   329→      setForm({
   330→        voucherDate: new Date().toISOString().slice(0, 10),
   331→        reference: '', bookAccountId: '', branchId: session?.branchId || branches[0]?.id || '',
   332→        description: '', lines: [emptyLine()],
   333→      })
   334→    }
   335→  }, [open])
   336→
   337→  // Tax calc: when amount or taxRate changes, auto-calc taxAmount = amount * taxRate / 100
   338→  const onLineAmountChange = (i: number, amt: number) => {
   339→    setForm((f: any) => ({
   340→      ...f,
   341→      lines: f.lines.map((l: any, idx: number) => {
   342→        if (idx !== i) return l
   343→        const taxRate = Number(l.taxRate) || 0
   344→        const taxAmount = l.taxAccountId ? Math.round(amt * taxRate) / 100 : 0
   345→        return { ...l, amount: amt, taxAmount }
   346→      }),
   347→    }))
   348→  }
   349→  const onLineTaxAccountChange = (i: number, taxAccountId: string) => {
   350→    setForm((f: any) => ({
   351→      ...f,
   352→      lines: f.lines.map((l: any, idx: number) => {
   353→        if (idx !== i) return l
   354→        const th = taxHeads.find((t: any) => t.id === taxAccountId)
   355→        const taxRate = th?.rate || 0
   356→        const amt = Number(l.amount) || 0
   357→        return { ...l, taxAccountId, taxRate, taxAmount: taxAccountId ? Math.round(amt * taxRate) / 100 : 0 }
   358→      }),
   359→    }))
   360→  }
   361→  const onLineTaxRateChange = (i: number, taxRate: number) => {
   362→    setForm((f: any) => ({
   363→      ...f,
   364→      lines: f.lines.map((l: any, idx: number) => {
   365→        if (idx !== i) return l
   366→        const amt = Number(l.amount) || 0
   367→        return { ...l, taxRate, taxAmount: l.taxAccountId ? Math.round(amt * taxRate) / 100 : 0 }
   368→      }),
   369→    }))
   370→  }
   371→
   372→  const setLine = (i: number, patch: any) => {
   373→    setForm((f: any) => ({ ...f, lines: f.lines.map((l: any, idx: number) => idx === i ? { ...l, ...patch } : l) }))
   374→  }
   375→  const addLine = () => setForm((f: any) => ({ ...f, lines: [...f.lines, emptyLine()] }))
   376→  const removeLine = (i: number) => setForm((f: any) => ({ ...f, lines: f.lines.filter((_: any, idx: number) => idx !== i) }))
   377→
   378→  // Totals
   379→  const totalDetailAmount = form.lines.reduce((s: number, l: any) => s + (Number(l.amount) || 0), 0)
   380→  const totalTaxAmount = form.lines.reduce((s: number, l: any) => s + (Number(l.taxAmount) || 0), 0)
   381→  const grandTotal = totalDetailAmount + totalTaxAmount
   382→  const totalDebitJV = form.lines.reduce((s: number, l: any) => s + (Number(l.debit) || 0), 0)
   383→  const totalCreditJV = form.lines.reduce((s: number, l: any) => s + (Number(l.credit) || 0), 0)
   384→  const balancedJV = Math.abs(totalDebitJV - totalCreditJV) < 0.01
   385→  const canSave = isJV ? balancedJV : (form.bookAccountId && form.lines.length > 0 && form.lines.every((l: any) => l.accountId) && totalDetailAmount > 0)
   386→
   387→  const save = async () => {
   388→    if (!form.voucherDate || !form.branchId) { toast.error('Date and branch are required'); return }
   389→    if (!isJV && !form.bookAccountId) { toast.error('Book Account is required'); return }
   390→    if (isJV && !balancedJV) { toast.error(`JV not balanced: Dr ${totalDebitJV} vs Cr ${totalCreditJV}`); return }
   391→    try {
   392→      const payload: any = {
   393→        voucherType: defaultType,
   394→        voucherDate: form.voucherDate,
   395→        branchId: form.branchId,
   396→        bookAccountId: isJV ? null : form.bookAccountId,
   397→        description: form.description,
   398→        reference: form.reference,
   399→        status: 'Posted',
   400→        lines: form.lines.map((l: any) => ({
   401→          accountId: l.accountId,
   402→          amount: Number(l.amount) || 0,
   403→          debit: isJV ? (Number(l.debit) || 0) : 0,
   404→          credit: isJV ? (Number(l.credit) || 0) : 0,
   405→          lineDescription: l.lineDescription,
   406→          taxAccountId: l.taxAccountId || null,
   407→          taxRate: Number(l.taxRate) || 0,
