   160→function useFetch<T>(url: string | null) {
   161→  const [data, setData] = useState<T | null>(null)
   162→  const [loading, setLoading] = useState(false)
   163→  const [error, setError] = useState<string | null>(null)
   164→  const reload = useCallback(async () => {
   165→    if (!url) return
   166→    setLoading(true)
   167→    setError(null)
   168→    try {
   169→      const res = await fetch(url)
   170→      if (res.status === 401) { setData(null); setError('Unauthorized'); return }
   171→      const json = await res.json()
   172→      if (!res.ok) setError(json.error || 'Failed')
   173→      else setData(json)
   174→    } catch (e: any) {
   175→      setError(e.message)
   176→    } finally {
   177→      setLoading(false)
   178→    }
   179→  }, [url])
   180→  useEffect(() => { reload() }, [reload])
   181→  return { data, loading, error, reload, setData }
   182→}

   184→async function apiPost(url: string, body: any) {
   185→  const res = await fetch(url, {
   186→    method: 'POST',
   187→    headers: { 'Content-Type': 'application/json' },
   188→    body: JSON.stringify(body),
   189→  })
   190→  const json = await res.json()
   191→  if (!res.ok) throw new Error(json.error || 'Failed')
   192→  return json
   193→}
   194→
   195→async function apiPatch(url: string, body: any) {
   196→  const res = await fetch(url, {
   197→    method: 'PATCH',
   198→    headers: { 'Content-Type': 'application/json' },
   199→    body: JSON.stringify(body),
   200→  })
   201→  const json = await res.json()
   202→  if (!res.ok) throw new Error(json.error || 'Failed')
   203→  return json
   204→}

   206→async function apiDelete(url: string) {
   207→  const res = await fetch(url, { method: 'DELETE' })
   208→  const json = await res.json()
   209→  if (!res.ok) throw new Error(json.error || 'Failed')
   210→  return json
   211→}

   216→function fmtMoney(n: number | null | undefined) {
   217→  if (n === null || n === undefined) return '—'
   218→  return Number(n).toLocaleString('en-US', { minimumFractionDigits: 0, maximumFractionDigits: 2 })
   219→}
   220→
   221→function fmtDateStr(d: string | Date | null | undefined) {
   222→  if (!d) return '—'
   223→  try { return fmtDate(new Date(d), 'dd MMM yyyy') } catch { return '—' }
   224→}
   225→
   226→function fmtDateTime(d: string | Date | null | undefined) {
   227→  if (!d) return '—'
   228→  try { return fmtDate(new Date(d), 'dd MMM yyyy HH:mm') } catch { return '—' }
   229→}

   234→export default function Home() {
   235→  const [companyName, setCompanyName] = useState('Contoura Gym')
   236→  const [session, setSession] = useState<SessionUser | null>(null)
   237→  const [loadingSession, setLoadingSession] = useState(true)
   238→  const [branches, setBranches] = useState<Branch[]>([])
   239→  const [selectedBranchIds, setSelectedBranchIds] = useState<string[]>([])
   240→  const [activeModule, setActiveModule] = useState<ModuleKey>('dashboard')
   241→  const [sidebarOpen, setSidebarOpen] = useState(false)
   242→
   243→  const refreshSession = useCallback(async () => {
   244→    setLoadingSession(true)
   245→    try {
   246→      const res = await fetch('/api/auth')
   247→      if (res.ok) {
   248→        const json = await res.json()
   249→        setSession(json.user)
   250→      } else {
   251→        setSession(null)
   252→      }
   253→    } catch {
   254→      setSession(null)
   255→    } finally {
   256→      setLoadingSession(false)
   257→    }
   258→  }, [])
   259→
   260→  const loadBranches = useCallback(async () => {
   261→    const res = await fetch('/api/branches')
   262→    if (res.ok) {
   263→      const json = await res.json()
   264→      setBranches(json.branches || [])
   265→    }
   266→  }, [])
   267→
   268→  useEffect(() => { refreshSession() }, [refreshSession])
   269→
   270→  useEffect(() => {
   271→    fetch('/api/company').then(r => r.json()).then(d => {
   272→      if (d.company?.name) setCompanyName(d.company.name)
   273→    }).catch(() => {})
   274→  }, [])
   275→
   276→  useEffect(() => {
   277→    if (session) {
   278→      loadBranches()
   279→    } else {
   280→      setActiveModule('login')
   281→    }
   282→  }, [session, loadBranches])
   283→
   284→  const login = useCallback(async (username: string, password: string) => {
   285→    try {
   286→      const res = await fetch('/api/auth', {
   287→        method: 'POST', headers: { 'Content-Type': 'application/json' },
   288→        body: JSON.stringify({ username, password }),
   289→      })
   290→      const json = await res.json()
   291→      if (!res.ok) throw new Error(json.error || 'Login failed')
   292→      await refreshSession()
   293→      return true
   294→    } catch (e: any) {
   295→      toast.error(e.message)
   296→      return false
   297→    }
   298→  }, [refreshSession])
   299→
   300→  const logout = useCallback(async () => {
   301→    await fetch('/api/auth', { method: 'DELETE' })
   302→    setSession(null)
   303→    setBranches([])
   304→    setSelectedBranchIds([])
   305→  }, [])
   306→
   307→  const has = useCallback((perm: string) => {
   308→    if (!session) return false
   309→    return session.permissions.includes(perm)
   310→  }, [session])
   311→
   312→  if (loadingSession) {
   313→    return (
   314→      <div className="min-h-screen flex items-center justify-center bg-background">
   315→        <div className="text-muted-foreground">Loading…</div>
   316→      </div>
   317→    )
   318→  }
   319→
   320→  if (!session) {
   321→    return <LoginScreen login={login} />
   322→  }
   323→
   324→  // Filter nav by permissions
   325→  const visibleNav = NAV.map(group => {
   326→    if ((group as any).key) return group
   327→    const children = (group as any).children.filter((c: any) => has(c.perm))
   328→    return children.length ? { ...group, children } : null
   329→  }).filter(Boolean) as any[]
   330→
   331→  return (
   332→    <AppContext.Provider value={{
   333→      session, branches, selectedBranchIds, setSelectedBranchIds,
   334→      has, refreshSession, logout, login,
   335→    }}>
   336→      <div className="min-h-screen bg-muted/30">
   337→        {/* Topbar */}
   338→        <header className="sticky top-0 z-30 h-14 bg-background border-b flex items-center px-3 sm:px-6 gap-2">
   339→          <Sheet open={sidebarOpen} onOpenChange={setSidebarOpen}>
   340→            <SheetTrigger asChild>
   341→              <Button variant="ghost" size="icon" className="lg:hidden">
   342→                <Menu className="h-5 w-5" />
   343→              </Button>
   344→            </SheetTrigger>
   345→            <SheetContent side="left" className="w-72 p-0">
   346→              <SidebarContent
   347→                nav={visibleNav}
   348→                active={activeModule}
   349→                onNavigate={(k) => { setActiveModule(k); setSidebarOpen(false) }}
   350→                session={session}
   351→                onLogout={logout}
   352→              />
   353→            </SheetContent>
   354→          </Sheet>
   355→
   356→          <div className="font-semibold text-lg flex items-center gap-2">
   357→            <div className="h-7 w-7 rounded bg-primary/15 text-primary flex items-center justify-center text-sm font-bold">C</div>
   358→            <span className="hidden sm:inline">{companyName}</span>
   359→            <span className="sm:hidden text-base">{companyName.slice(0, 8)}</span>
   360→          </div>
   361→
   362→          <div className="flex-1" />
   363→
   364→          <BranchSelector
   365→            branches={branches}
   366→            selected={selectedBranchIds}
   367→            onChange={setSelectedBranchIds}
   368→          />
   369→
   370→          <div className="hidden md:flex items-center gap-2 px-3 py-1.5 rounded border bg-muted/40">
   371→            <div className="h-7 w-7 rounded-full bg-primary/15 text-primary flex items-center justify-center text-xs font-bold">
   372→              {session.fullName.charAt(0).toUpperCase()}
   373→            </div>
   374→            <div className="text-xs leading-tight">
   375→              <div className="font-medium">{session.fullName}</div>
   376→              <div className="text-muted-foreground">{session.roleName}</div>
   377→            </div>
   378→          </div>
   379→
   380→          <Button variant="ghost" size="icon" onClick={logout} title="Logout">
   381→            <LogOut className="h-4 w-4" />
   382→          </Button>
   383→        </header>
   384→
   385→        <div className="flex">
   386→          {/* Desktop Sidebar */}
   387→          <aside className="hidden lg:block w-60 border-r bg-background min-h-[calc(100vh-3.5rem)] sticky top-14">
   388→            <SidebarContent
   389→              nav={visibleNav}
   390→              active={activeModule}
   391→              onNavigate={setActiveModule}

   479→function BranchSelector({ branches, selected, onChange }: any) {
   480→  const [open, setOpen] = useState(false)
   481→  const all = selected.length === 0
   482→  return (
   483→    <div className="relative">
   484→      <Button variant="outline" size="sm" onClick={() => setOpen(o => !o)} className="gap-1.5 max-w-[180px]">
   485→        <Building2 className="h-3.5 w-3.5" />
   486→        <span className="truncate">{all ? 'All branches' : `${selected.length} selected`}</span>
   487→        <ChevronDown className="h-3 w-3" />
   488→      </Button>
   489→      {open && (
   490→        <>
   491→          <div className="fixed inset-0 z-40" onClick={() => setOpen(false)} />
   492→          <div className="absolute right-0 mt-1 w-64 bg-background border rounded shadow-lg z-50 max-h-80 overflow-y-auto">
   493→            <div className="p-2 border-b flex items-center justify-between">
   494→              <span className="text-sm font-medium">Filter branches</span>
   495→              <button onClick={() => onChange([])} className="text-xs text-primary hover:underline">All</button>
   496→            </div>
   497→            {branches.map((b: Branch) => (
   498→              <label key={b.id} className="flex items-center gap-2 px-3 py-1.5 hover:bg-muted cursor-pointer text-sm">
   499→                <input
   500→                  type="checkbox"

   687→// Opening Trial Balance — COA grid with Dr/Cr, allows unbalanced save
   688→// =================================================================
   689→function OpeningTrialBalanceScreen() {
   690→  const { has } = useApp()
   691→  const [entries, setEntries] = useState<Record<string, { debit: number; credit: number }>>({})
   692→  const [saving, setSaving] = useState(false)
   693→
   694→  const { data, reload } = useFetch<any>('/api/opening-balance')
   695→
   696→  useEffect(() => {
   697→    if (data?.accounts) {
   698→      const map: Record<string, { debit: number; credit: number }> = {}
   699→      for (const a of data.accounts) {
   700→        map[a.id] = { debit: a.debit || 0, credit: a.credit || 0 }
   701→      }
   702→      setEntries(map)
   703→    }
   704→  }, [data])
   705→
   706→  const accounts = data?.accounts || []
   707→  const hasExisting = data?.hasExisting || false
   708→
   709→  const totalDebit = Object.values(entries).reduce((s, e) => s + (Number(e.debit) || 0), 0)
   710→  const totalCredit = Object.values(entries).reduce((s, e) => s + (Number(e.credit) || 0), 0)
   711→  const difference = totalCredit - totalDebit
   712→
   713→  const setDr = (id: string, val: number) => {
   714→    if (val < 0) val = 0
   715→    setEntries(prev => ({ ...prev, [id]: { debit: val, credit: 0 } }))
   716→  }
   717→  const setCr = (id: string, val: number) => {
   718→    if (val < 0) val = 0
   719→    setEntries(prev => ({ ...prev, [id]: { debit: 0, credit: val } }))
   720→  }
   721→
   722→  const save = async () => {
   723→    // NO balance requirement — OTB is allowed to save unbalanced
   724→    setSaving(true)
   725→    try {
   726→      const entryArray = Object.entries(entries).map(([id, v]) => ({ id, debit: Number(v.debit) || 0, credit: Number(v.credit) || 0 }))
   727→      await apiPost('/api/opening-balance', { entries: entryArray })
   728→      toast.success(hasExisting ? 'Opening balances updated' : 'Opening balances saved')
   729→      reload()
   730→    } catch (e: any) {
   731→      toast.error(e.message)
   732→    } finally {
   733→      setSaving(false)
   734→    }
   735→  }
   736→
   737→  return (
   738→    <div>
   739→      <PageHeader title="Opening Trial Balance"
   740→        action={has('vouchers.add') ? save : undefined}
   741→        actionLabel={saving ? 'Saving…' : (hasExisting ? 'Save Changes' : 'Save')}
   742→      />
   743→      <div className="text-xs text-muted-foreground mb-3">
   744→        Enter opening amounts against detail accounts. Debit OR Credit per account. Unbalanced entries are allowed.
   745→      </div>
   746→      <div className="border rounded overflow-x-auto max-h-[65vh] overflow-y-auto">
   747→        <table className="w-full text-sm">
   748→          <thead className="bg-muted/50 border-b sticky top-0 z-10">
   749→            <tr>
   750→              <th className="px-3 py-2 text-left font-medium">Code</th>
   751→              <th className="px-3 py-2 text-left font-medium">Account Name</th>
   752→              <th className="px-3 py-2 text-left font-medium">Type</th>
   753→              <th className="px-3 py-2 text-right font-medium w-32">Debit</th>
   754→              <th className="px-3 py-2 text-right font-medium w-32">Credit</th>
   755→            </tr>
   756→          </thead>
   757→          <tbody>
   758→            {accounts.map((a: any) => (
   759→              <tr key={a.id} className={"border-b last:border-0 hover:bg-muted/30 " + (!a.isActive ? 'opacity-50' : '')}>
   760→                <td className="px-3 py-1.5 font-mono text-xs">{a.code}</td>
   761→                <td className="px-3 py-1.5">{a.name}{!a.isActive && <span className="ml-1 text-xs text-muted-foreground">(inactive)</span>}</td>
   762→                <td className="px-3 py-1.5 text-xs">{a.accountType}</td>
   763→                <td className="px-3 py-1.5"><Input type="number" min={0} step="0.01" value={entries[a.id]?.debit || ''} onChange={e => setDr(a.id, Number(e.target.value))} className="h-7 text-right" placeholder="0" /></td>
   764→                <td className="px-3 py-1.5"><Input type="number" min={0} step="0.01" value={entries[a.id]?.credit || ''} onChange={e => setCr(a.id, Number(e.target.value))} className="h-7 text-right" placeholder="0" /></td>
   765→              </tr>
   766→            ))}
   767→            {accounts.length === 0 && <tr><td colSpan={5} className="px-3 py-8 text-center text-muted-foreground">No detail accounts found in COA.</td></tr>}
   768→          </tbody>
   769→        </table>
   770→      </div>
   771→      <div className="sticky bottom-0 mt-3 bg-background border rounded p-3 flex items-center gap-6 shadow">
   772→        <div className="text-sm"><span className="text-muted-foreground">Total Debit: </span><span className="font-mono font-semibold">{fmtMoney(totalDebit)}</span></div>
   773→        <div className="text-sm"><span className="text-muted-foreground">Total Credit: </span><span className="font-mono font-semibold">{fmtMoney(totalCredit)}</span></div>
   774→        <div className="text-sm"><span className="text-muted-foreground">Difference (Cr − Dr): </span><span className={"font-mono font-semibold " + (Math.abs(difference) < 0.01 ? 'text-green-600' : 'text-amber-600')}>{difference > 0 ? '+' : ''}{fmtMoney(difference)}</span></div>
   775→        <div className="flex-1" />
   776→        <div className={"text-xs font-medium " + (Math.abs(difference) < 0.01 ? 'text-green-600' : 'text-amber-600')}>{Math.abs(difference) < 0.01 ? 'Balanced' : 'Unbalanced (save allowed)'}</div>
   777→      </div>
   778→    </div>
   779→  )
   780→}
